cmake_minimum_required(VERSION 3.27)
# VERSION at the repository root; the shipped rebuild sources (/usr/src/grassfield-compositor) carry their own cmake/ directory.
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/cmake/GrassfieldVersion.cmake)
  include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/GrassfieldVersion.cmake)
else()
  include(${CMAKE_CURRENT_SOURCE_DIR}/../cmake/GrassfieldVersion.cmake)
endif()
project(grassfield-compositor VERSION ${GF_VERSION_NUMERIC} LANGUAGES CXX)
add_compile_definitions(GRASSFIELD_VERSION="${GF_VERSION}")

# Build against the INSTALLED Hyprland headers (master-prompt §5.6). Outputs:
#   build/grassfield-compositor.so      the plugin
#   build/grassfield-compositor-tests   unit tests for src/core (no Hyprland needed)
# Interim equivalent without CMake: build-direct.sh.

set(CMAKE_CXX_STANDARD 23) # Hyprland 0.56.2 headers use std::expected and string_view::contains
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON) # for clang-tidy (gate Q1)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

# Where the plugin and the tests are written: compositor/build by default (just plugin-build, the PKGBUILD); the rebuild
# on the target (packaging/arch/grassfield-compositor-rebuild, ADR-0009 amendment) builds from the read-only sources in
# /usr/src/grassfield-compositor into a private directory.
set(GRASSFIELD_OUTPUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/build" CACHE PATH "Output directory of the plugin and the unit tests")

find_package(PkgConfig REQUIRED)
pkg_check_modules(HYPRLAND REQUIRED IMPORTED_TARGET hyprland pixman-1 libdrm glesv2) # glesv2: the Genie draw (GenieDraw.cpp)
pkg_check_modules(LUA REQUIRED lua) # headers only: symbols resolve from Hyprland's liblua at load time

# Pure logic, no Hyprland includes (unit-testable).
add_library(gfcore STATIC src/core/Protocol.cpp src/core/TitlebarLayout.cpp src/core/PointerGesture.cpp src/core/Minimize.cpp src/core/Appmenu.cpp
  src/core/EventThrottle.cpp src/core/Overview.cpp src/core/DesktopFocus.cpp src/core/Json.cpp src/core/Stages.cpp src/core/CursorRefresh.cpp
  src/core/DeferredFocus.cpp src/core/OverviewFocus.cpp src/core/Genie.cpp src/core/Tiling.cpp src/core/LayerPointer.cpp src/core/ClientChrome.cpp src/core/FrameLog.cpp src/core/Quiet.cpp
  src/core/Notices.cpp src/core/RunMode.cpp src/core/TopClamp.cpp src/core/WindowRim.cpp src/core/CapturePrivacy.cpp)
target_include_directories(gfcore PUBLIC src)
set_target_properties(gfcore PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(gfcore PRIVATE -Wall -Wextra)

# Wayland protocol server code generated by hyprwayland-scanner (org_kde_kwin_appmenu v2), as in build-direct.sh.
find_program(HYPRWAYLAND_SCANNER hyprwayland-scanner REQUIRED)
set(GF_PROTO_DIR ${CMAKE_CURRENT_BINARY_DIR}/protocols)
add_custom_command(
  OUTPUT ${GF_PROTO_DIR}/appmenu.cpp ${GF_PROTO_DIR}/appmenu.hpp
  COMMAND ${CMAKE_COMMAND} -E make_directory ${GF_PROTO_DIR}
  COMMAND ${HYPRWAYLAND_SCANNER} ${CMAKE_CURRENT_SOURCE_DIR}/protocols/appmenu.xml ${GF_PROTO_DIR}
  DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/protocols/appmenu.xml)
add_library(gfprotocols OBJECT ${GF_PROTO_DIR}/appmenu.cpp)
target_link_libraries(gfprotocols PRIVATE PkgConfig::HYPRLAND)
set_target_properties(gfprotocols PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)

add_library(grassfield-compositor SHARED src/main.cpp src/Titlebar.cpp src/Ipc.cpp src/Minimize.cpp src/AppmenuServer.cpp src/Overview.cpp src/DesktopFocus.cpp src/Surfaces.cpp src/WindowTransform.cpp src/Stages.cpp src/CursorRefresh.cpp src/DeferredFocus.cpp src/OverviewFocus.cpp src/GenieDraw.cpp src/Tiling.cpp src/LayerPointer.cpp src/ClientChrome.cpp src/FrameProbe.cpp src/TopClamp.cpp src/WindowRim.cpp src/CapturePrivacy.cpp $<TARGET_OBJECTS:gfprotocols>)
target_link_libraries(grassfield-compositor PRIVATE gfcore PkgConfig::HYPRLAND)
target_include_directories(grassfield-compositor PRIVATE ${LUA_INCLUDE_DIRS} ${CMAKE_CURRENT_BINARY_DIR})
target_compile_options(grassfield-compositor PRIVATE -Wall -Wextra)
set_target_properties(grassfield-compositor PROPERTIES
  PREFIX ""
  LIBRARY_OUTPUT_DIRECTORY ${GRASSFIELD_OUTPUT_DIR})

# The plugin's ABI stamp, next to it, from the version.h it is compiled against (ADR-0009 "Amendment 2026-10-23"): the
# package installs it with the plugin, the user rebuild with its build, and the session loads a build only when its stamp
# is the running Hyprland's ABI. Regenerated whenever that header changes, like the plugin.
find_file(GF_HYPRLAND_VERSION_H NAMES hyprland/src/version.h HINTS ${HYPRLAND_INCLUDE_DIRS} REQUIRED)
add_custom_command(
  OUTPUT ${GRASSFIELD_OUTPUT_DIR}/grassfield-compositor.abi
  COMMAND ${CMAKE_COMMAND} -DVERSION_H=${GF_HYPRLAND_VERSION_H} -DOUT=${GRASSFIELD_OUTPUT_DIR}/grassfield-compositor.abi
          -P ${CMAKE_CURRENT_SOURCE_DIR}/abi-stamp.cmake
  DEPENDS ${GF_HYPRLAND_VERSION_H} ${CMAKE_CURRENT_SOURCE_DIR}/abi-stamp.cmake
  VERBATIM)
add_custom_target(grassfield-compositor-abi ALL DEPENDS ${GRASSFIELD_OUTPUT_DIR}/grassfield-compositor.abi)
# Building the plugin alone (cmake --build --target grassfield-compositor, just plugin-build) refreshes its stamp too.
add_dependencies(grassfield-compositor grassfield-compositor-abi)

option(GRASSFIELD_BUILD_TESTS "Build unit tests" ON)
if(GRASSFIELD_BUILD_TESTS)
  find_package(GTest REQUIRED)
  enable_testing()
  add_executable(grassfield-compositor-tests tests/ProtocolTest.cpp tests/TitlebarLayoutTest.cpp tests/PointerGestureTest.cpp tests/MinimizeTest.cpp tests/AppmenuTest.cpp
    tests/EventThrottleTest.cpp tests/OverviewTest.cpp tests/DesktopFocusTest.cpp tests/StagesTest.cpp tests/CursorRefreshTest.cpp
    tests/DeferredFocusTest.cpp tests/OverviewFocusTest.cpp tests/GenieTest.cpp tests/TilingTest.cpp tests/LayerPointerTest.cpp tests/ClientChromeTest.cpp tests/FrameLogTest.cpp tests/QuietTest.cpp
    tests/NoticesTest.cpp tests/RunModeTest.cpp tests/TopClampTest.cpp tests/WindowRimTest.cpp tests/CapturePrivacyTest.cpp tests/ParserFuzzTest.cpp)
  target_link_libraries(grassfield-compositor-tests PRIVATE gfcore GTest::gtest_main)
  # QuietTest reads the plugin's sources (every key listener opens the quiet scope).
  # RunModeTest reads the run-mode vectors shared with the shell and the apps (ADR-0019).
  target_compile_definitions(grassfield-compositor-tests PRIVATE GF_PLUGIN_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/src"
    GF_RUN_MODE_VECTORS="${CMAKE_CURRENT_SOURCE_DIR}/../tests/fixtures/run-mode-vectors.json")
  target_compile_options(grassfield-compositor-tests PRIVATE -Wall -Wextra)
  set_target_properties(grassfield-compositor-tests PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${GRASSFIELD_OUTPUT_DIR})
  add_test(NAME grassfield-compositor-tests COMMAND grassfield-compositor-tests)
endif()
